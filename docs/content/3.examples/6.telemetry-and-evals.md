An invoice total looks wrong. You want to see what was asked, which provider attempt ran, and what the application actually stored. Those are related pieces of evidence, but they don't all come from the same place.

In this example, we'll capture a provider attempt, keep the command's prompt and parameters with its output reference, then read and grade the stored JSON. Everything runs locally with a fake client.

This example extends the [quick start](/getting-started/quick-start). Keep its `ExtractInvoice.cs` and local NuGet feed. Add `Zeeq.LlmCmd.Providers` version `0.1.0-alpha.1` to the project, add the helper files below, and replace `Program.cs` with the execution listing. The last section contains the offline-only helpers.

## Set up attempt capture

This helper registers a keyed MEAI client with provider extraction at its inner boundary. The sink keeps observations in memory for this example. A real sink can forward normalized snapshots to a host-owned destination.

```csharp [CaptureSetup.cs]
using System.Collections.Concurrent;
using System.Text.Json;
using Microsoft.Extensions.AI;
using Microsoft.Extensions.DependencyInjection;
using Zeeq.LlmCmd.Providers;
using Zeeq.LlmCmd.Providers.Platforms;

namespace QuickStart;

public static class CaptureSetup
{
    // The identity selects the protocol adapter; it is not inferred from a model.
    private static readonly ProviderIdentity Identity = new(
        "azureopenai",
        "azure",
        ApiKind.ChatCompletions
    );

    public static ProviderExtractor CreateExtractor(
        IEnumerable<INativeResponseProjector>? projectors = null
    )
    {
        IProviderResponseAdapter[] adapters = [new OpenAiChatAdapter("azureopenai")];
        return new ProviderExtractor(new ProviderAdapterRegistry(adapters, projectors));
    }

    public static void AddClient(
        IServiceCollection services,
        Func<IServiceProvider, IChatClient> createProvider,
        IExtractionSink sink,
        IEnumerable<INativeResponseProjector>? projectors = null
    )
    {
        services
            .AddKeyedChatClient("azure", createProvider)
            .UseProviderExtraction(CreateExtractor(projectors), Identity, sink);
    }

    // Manual capture is useful when the host already has a completed response.
    public static ExtractionResult CaptureExisting(
        ChatResponse response,
        string attemptId,
        JsonElement? wire = null
    ) => CreateExtractor().Extract(Identity, response, attemptId, wire);
}

public sealed class AttemptSink : IExtractionSink
{
    private readonly ConcurrentQueue<ExtractionResult> results = new();

    public ExtractionResult[] Results => results.ToArray();

    public ValueTask OnExtractedAsync(
        ExtractionResult result,
        CancellationToken cancellationToken
    )
    {
        results.Enqueue(result);
        return ValueTask.CompletedTask;
    }
}
```

## Define an input record for later evaluation

This is application code, not a built-in storage interface. It pairs the captured command snapshot with the exact output reference. The application chooses where and how to store it.

```csharp [EvaluationCapture.cs]
using Zeeq.LlmCmd;

namespace QuickStart;

public sealed record EvaluationInput(
    Guid ExecutionId,
    ExecutionSnapshot Snapshot,
    OutputRef Output
);

public interface IEvaluationInputStore
{
    ValueTask SaveAsync(EvaluationInput input, CancellationToken cancellationToken);
}

public static class EvaluationCapture
{
    public static ValueTask SaveAsync<T>(
        Outcome<T>.Success success,
        IEvaluationInputStore store,
        CancellationToken cancellationToken = default
    )
    {
        // A valid value alone is not proof that there is an output to read later.
        if (success.Delivery is not OutputDelivery.Stored stored)
        {
            throw new InvalidOperationException("The output was not stored.");
        }

        return store.SaveAsync(
            new EvaluationInput(
                success.ExecutionId,
                success.Snapshot,
                stored.Output.Reference
            ),
            cancellationToken
        );
    }
}
```

## Execute, inspect, and grade

```csharp [TelemetryProgram.cs]
using System.Collections.Concurrent;
using System.Diagnostics;
using Microsoft.Extensions.AI;
using Microsoft.Extensions.DependencyInjection;
using QuickStart;
using Zeeq.LlmCmd;
using Zeeq.LlmCmd.Generated;

var activities = new ConcurrentQueue<string>();

// A listener must be present when the command starts to record its activities.
using var listener = new ActivityListener
{
    ShouldListenTo = source => source.Name == "Zeeq.LlmCmd",
    Sample = (ref ActivityCreationOptions<ActivityContext> _) =>
        ActivitySamplingResult.AllData,
    ActivityStopped = activity => activities.Enqueue(activity.OperationName),
};
ActivitySource.AddActivityListener(listener);

var actor = new Actor("demo", "reviewer");
var sink = new AttemptSink();
var client = new SyntheticClient();
var services = new ServiceCollection();

// The provider wrapper observes attempts without changing the command call site.
CaptureSetup.AddClient(services, _ => client, sink);
services
    .AddLlmCmd(options =>
        options
            .AddChatClient("azure")
            .AddProfile("extract", "azure", "synthetic-model")
            .UseActor(_ => actor)
    )
    .AddGeneratedCommands();

using var provider = services.BuildServiceProvider();
var command = provider.GetRequiredService<ExtractInvoice>();
var ledger = provider.GetRequiredService<IHumanFeedbackLedger>();
var outcome = await command
    .WithParameters(new InvoiceInput("Invoice for USD 12.50"))
    .ExecuteAsync();

if (
    outcome
    is not Outcome<Invoice>.Success { Delivery: OutputDelivery.Stored stored } success
)
{
    throw new InvalidOperationException("The invoice was not stored.");
}

var attempt = sink.Results.Single().Snapshot;
var outputRef = stored.Output.Reference;

// Keep prompt inputs linked to the version the ledger accepted.
var inputs = new InMemoryEvaluationInputs();
await EvaluationCapture.SaveAsync(success, inputs);
var retained = inputs.Items.Single();
if (
    retained.Output != outputRef
    || retained.Snapshot.RawTemplate != "Extract the invoice from {Document}."
    || retained.Snapshot.Parameters.Single().Name != "Document"
)
{
    throw new InvalidOperationException("The evaluation input was not captured.");
}

if (
    ledger.ReadOutput(outputRef, actor)
    is not LedgerResult<OutputArtifact>.Accepted read
)
{
    throw new InvalidOperationException("The stored output could not be read.");
}

// Grade the value read from storage, not a separately reconstructed result.
var observedTotal = read.Value.Value.GetProperty("Total");
if (observedTotal.GetDecimal() != 12.50m || attempt.Usage.Input != 7)
{
    throw new InvalidOperationException(
        "The synthetic invoice did not pass evaluation."
    );
}

var grade = ledger.RecordGrade(
    new GradeRequest(
        Guid.NewGuid(),
        outputRef,
        "/Total",
        observedTotal,
        1.0,
        "Expected total"
    ),
    actor
);
if (grade is not LedgerResult<GradeReceipt>.Accepted)
{
    throw new InvalidOperationException("The grade was not recorded.");
}

// This second extraction illustrates the manual API. Do not publish it again
// as a new attempt unless the host deliberately handles duplicate observations.
var manual = CaptureSetup.CaptureExisting(client.LastResponse!, "manual-example");
if (manual.Snapshot.Usage.Input != 7 || !activities.Contains("llmcmd.execute"))
{
    throw new InvalidOperationException("Capture did not observe the expected work.");
}

// These inputs and outputs are synthetic. Apply a data policy to real records.
Console.WriteLine(
    $"Prompt: {success.Snapshot.PromptId}; input tokens: {attempt.Usage.Input}"
);
Console.WriteLine($"Template: {retained.Snapshot.RawTemplate}");
foreach (var parameter in retained.Snapshot.Parameters)
{
    Console.WriteLine($"{parameter.Name}: {parameter.Value}");
}
Console.WriteLine($"Stored output: {read.Value.Value.GetRawText()}");
Console.WriteLine("The exact stored total was read and graded.");
```

The example listens to command activities directly. It also captures one provider snapshot through middleware, saves the prompt snapshot with its output reference, and reads the exact output accepted by the ledger. `CaptureExisting` shows the alternative for a response the host already has.

## Offline helpers

Add this file to run the example. The fake client returns a fixed invoice and usage counts; the input store keeps the captured record in memory. Neither contacts a service.

::code-collapse{label="Offline helpers"}

```csharp [TelemetryFakes.cs]
using Microsoft.Extensions.AI;
using QuickStart;

// This store lasts only as long as the process. Replace it for durable records.
sealed class InMemoryEvaluationInputs : IEvaluationInputStore
{
    public List<EvaluationInput> Items { get; } = [];

    public ValueTask SaveAsync(
        EvaluationInput input,
        CancellationToken cancellationToken
    )
    {
        Items.Add(input);
        return ValueTask.CompletedTask;
    }
}

// A deterministic response makes this example safe to run without a provider.
sealed class SyntheticClient : IChatClient
{
    public ChatResponse? LastResponse { get; private set; }

    public Task<ChatResponse> GetResponseAsync(
        IEnumerable<ChatMessage> messages,
        ChatOptions? options = null,
        CancellationToken cancellationToken = default
    )
    {
        LastResponse = new ChatResponse(
            new ChatMessage(ChatRole.Assistant, """{"Total":12.50,"Currency":"USD"}""")
        )
        {
            ModelId = "synthetic-model",
            Usage = new()
            {
                InputTokenCount = 7,
                OutputTokenCount = 3,
                TotalTokenCount = 10,
            },
        };
        return Task.FromResult(LastResponse);
    }

    public IAsyncEnumerable<ChatResponseUpdate> GetStreamingResponseAsync(
        IEnumerable<ChatMessage> messages,
        ChatOptions? options = null,
        CancellationToken cancellationToken = default
    ) => throw new NotSupportedException();

    public object? GetService(Type serviceType, object? serviceKey = null) => null;

    public void Dispose() { }
}
```

::

The fake provides normalized MEAI usage but no native provider wire data, so extraction may be partial.

The example's input store and the default ledger are in memory. The ledger does not automatically persist the command snapshot with the output. A host that needs both after restart must implement durable storage and decide how to handle a failed input-record write. See [Telemetry and evaluation capture](/guides/accounting-and-observability) for exporter setup, sensitive-data boundaries, and whole-output evaluation.
